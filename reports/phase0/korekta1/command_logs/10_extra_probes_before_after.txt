## Script: extra_probes.py
"""Additional probes beyond the audit: related input-type and context gaps."""

import json
from datetime import date
from decimal import Decimal, Inexact, Rounded, localcontext

from market_intelligence.core.models import Candle, DataQuality, Instrument, MarketTick

results = []


def check(name, fn):
    try:
        results.append({"probe": name, "result": repr(fn())})
    except Exception as exc:
        results.append({"probe": name, "exception": type(exc).__name__, "msg": str(exc)[:160]})


ik = dict(
    exchange="BYBIT", symbol="BTCUSDT", market_type="USDT_LINEAR_PERPETUAL", base_asset="BTC",
    quote_asset="USDT", status="TRADING", tick_size="0.10",
)
i = Instrument(**ik)

for name, value in [
    ("bytearray_epoch", bytearray(b"1767225600")),
    ("bytes_iso", b"2026-01-01T00:00:00Z"),
    ("memoryview_iso", memoryview(b"2026-01-01T00:00:00Z")),
    ("date_object", date(2026, 1, 1)),
    ("str_epoch_with_Z", "1767225600Z"),
    ("str_padded_iso", " 2026-01-01T00:00:00Z "),
    ("str_date_only", "2026-01-01"),
    ("str_epoch_ms_float", "1767225600000.5"),
]:
    check(f"listed_at:{name}", lambda v=value: Instrument(**(ik | {"listed_at": v})).listed_at)

for name, value in [
    ("bytes_decimal", b"0.5"),
    ("decimal_one_schema", Decimal(1)),
]:
    field = "schema_version" if "schema" in name else "tick_size"
    check(f"{field}:{name}", lambda v=value, f=field: getattr(Instrument(**(ik | {f: v})), f))

check("symbol:bytes", lambda: Instrument(**(ik | {"symbol": b"BTCUSDT"})).symbol)

for sep in ["\x85", " ", " ", "\x0b", "\x0c", "\x1e", "\r\n"]:
    check(f"reason:{sep!r}", lambda s=sep: DataQuality(status="GAP", reason=f"a{s}b").reason)


def candle_under(ctx_kwargs, traps=()):
    with localcontext(**ctx_kwargs) as ctx:
        for trap in traps:
            ctx.traps[trap] = True
        return Candle(
            instrument=i, interval="1m", start="2026-01-05T12:00:00Z", end="2026-01-05T12:01:00Z",
            open="50000.123456789012345678901234567", high="50000.123456789012345678901234567",
            low="50000.123456789012345678901234567", close="50000.123456789012345678901234567",
            turnover_usdt="0", is_closed=True, source="SYNTHETIC", quality=DataQuality.valid(),
        ).open


check("candle_prec1", lambda: candle_under({"prec": 1}))
check("candle_traps_inexact_rounded", lambda: candle_under({"prec": 5}, (Inexact, Rounded)))
check("candle_emax_small", lambda: candle_under({"Emax": 2, "Emin": -2}))

bad = Instrument.model_construct(**(ik | {"tick_size": Decimal("-1")}))
check(
    "model_construct_nested",
    lambda: MarketTick(
        instrument=bad, last_price="1", received_at="2026-01-05T12:00:00Z", source="SYNTHETIC",
        quality=DataQuality.valid(),
    ).instrument.tick_size,
)

print(json.dumps(results, indent=2, ensure_ascii=False))
