## Script: k2_probes.py
"""Correction 2 probes: K1-01 and related timestamp text forms, plus the DEC-020 product case.

Every timestamp candidate is validated through Instrument.listed_at on the Python path and
the JSON path. SYNTHETIC data only.
"""

import json
import sys

from market_intelligence.core.models import DataQuality, Instrument, Trade

sys.stdout.reconfigure(encoding="utf-8")  # type: ignore[union-attr]

IK = dict(
    exchange="BYBIT", symbol="BTCUSDT", market_type="USDT_LINEAR_PERPETUAL", base_asset="BTC",
    quote_asset="USDT", status="TRADING", tick_size="0.10",
)
CANDIDATES = [
    # K1-01 (audit 2) and exponent variants found in correction 2
    ".5", "+.5", "-.5", ".5e3", ".5E+3", "-.5e-1",
    # text forms outside the RFC 3339 profile
    "2026-01-01 00:00:00+00:00", "2026-01-01T00:00:00z", "2026-01-01T00:00:00+0100",
    "2026-01-01T00:00Z", "2026-01-01T00:00:00.123456789Z", "2026-01-01T24:00:00Z",
    "2026-01-01T00:00:60Z", "20260101T000000Z", "2026-01-01",
    # allowed RFC 3339 profile
    "2026-01-01T00:00:00Z", "2026-01-01T00:00:00.5Z", "2026-01-01T00:00:00.123456Z",
    "2026-01-01T01:00:00+01:00", "2025-12-31T19:00:00-05:00",
    # naive (no offset)
    "2026-01-01T00:00:00",
]

results = []
for text in CANDIDATES:
    row = {"input": text}
    for mode in ("python", "json"):
        try:
            if mode == "python":
                value = Instrument.model_validate(IK | {"listed_at": text}).listed_at
            else:
                value = Instrument.model_validate_json(json.dumps(IK | {"listed_at": text})).listed_at
            row[mode] = "ACCEPTED " + value.isoformat()
        except Exception as exc:  # noqa: BLE001 - probe records every outcome
            row[mode] = f"{type(exc).__name__}: " + str(exc).splitlines()[2].strip()[:90]
    results.append(row)

trade = dict(
    instrument=Instrument(**IK), trade_id="synthetic-k2", price="0.0000000001",
    quantity="0.0000000001", value_usdt="0.00000000000000000001", aggressor_side="BUY",
    executed_at="2026-01-05T12:00:00Z", received_at="2026-01-05T12:00:00Z", source="SYNTHETIC",
    quality=DataQuality.valid(),
)
try:
    product = str(Trade(**trade).value_usdt)
    results.append({"input": "Trade 1E-10 x 1E-10 = 1E-20", "python": "ACCEPTED " + product})
except Exception as exc:  # noqa: BLE001
    results.append(
        {"input": "Trade 1E-10 x 1E-10 = 1E-20", "python": "ValidationError: " + str(exc).splitlines()[2].strip()[:90]}
    )

print(json.dumps(results, indent=2, ensure_ascii=False))
